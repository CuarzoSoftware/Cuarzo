250 // Fence-tracked deferred destruction: cleanup runs once its fence signals (drained by
251 // clearGarbage()). Lets submissions be async without freeing in-flight resources. Each entry
252 // records the thread that created the resources; cleanup MUST run on that same thread because
253 // Vulkan command pools require external synchronization (NVIDIA crashes on cross-thread
254 // destruction that Mesa tolerates). Hence a list per thread (in its Stream), drained by its
255 // clearGarbage(), and one for entries any thread can drain (m_anyGarbage).
256
257 // Shared by the entries of a batch, destroyed with the last one
258 struct Fence
259 {
260 Fence(VkDevice d, VkFence f) noexcept;
261 ~Fence() { if (fence != VK_NULL_HANDLE) vkDestroyFence(device, fence, nullptr); }
262 VkDevice device;
263 VkFence fence; // VK_NULL_HANDLE: already done
264 UInt64 serial; // Creation order, ~submission order (see GarbageList::blocking)
265 bool signaled { false }; // Cached, queried until it signals
266 };
