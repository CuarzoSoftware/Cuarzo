130 static std::array<Float32, 4> CalcUVBounds(const SkMatrix &uvMatrix, const SkIRect &dst,
131 const std::optional<SkRect> &validRect, SkISize imageSize) noexcept
132 {
133 SkRect area { SkRect::Make(dst) };
134
135 if (validRect)
136 {
137 SkRect valid { *validRect };
138
139 if (valid.intersect(area))
140 area = valid;
141 }
142
143 SkRect uv;
144 uvMatrix.mapRect(&uv, area);
145 uv.sort();
146 const Float32 hx { 1.5f / Float32(std::max(1, imageSize.width())) };
147 const Float32 hy { 1.5f / Float32(std::max(1, imageSize.height())) };
148 uv.inset(std::min(hx, uv.width() * 0.5f), std::min(hy, uv.height() * 0.5f));
149 return { uv.left(), uv.top(), uv.right(), uv.bottom() };
150 }

157 static std::array<Float32, 12> MakeVibrancyMatrix(Float32 saturation, const SkColor4f &tint) noexcept
158 {
159 constexpr Float32 w[3] { 0.2126f, 0.7152f, 0.0722f };
160 const Float32 a { std::clamp(tint.fA, 0.f, 1.f) };
161 const Float32 t[3] { tint.fR, tint.fG, tint.fB };
162 std::array<Float32, 12> m {};
163
164 for (int i = 0; i < 3; i++)
165 {
166 for (int j = 0; j < 3; j++)
167 m[i * 4 + j] = (1.f - a) * ((1.f - saturation) * w[j] + (i == j ? saturation : 0.f));
168
169 m[i * 4 + 3] = a * t[i];
170 }
171
172 return m;
173 }
