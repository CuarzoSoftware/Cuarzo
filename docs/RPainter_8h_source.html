140 static std::array<Float32, 4> CalcUVBounds(const SkMatrix &uvMatrix, const SkIRect &dst,
141 const std::optional<SkRect> &validRect, SkISize imageSize) noexcept
142 {
143 SkRect area { SkRect::Make(dst) };
144
145 if (validRect)
146 {
147 SkRect valid { *validRect };
148
149 if (valid.intersect(area))
150 area = valid;
151 }
152
153 SkRect uv;
154 uvMatrix.mapRect(&uv, area);
155 uv.sort();
156 const Float32 hx { 1.5f / Float32(std::max(1, imageSize.width())) };
157 const Float32 hy { 1.5f / Float32(std::max(1, imageSize.height())) };
158 uv.inset(std::min(hx, uv.width() * 0.5f), std::min(hy, uv.height() * 0.5f));
159 return { uv.left(), uv.top(), uv.right(), uv.bottom() };
160 }

167 static std::array<Float32, 12> MakeVibrancyMatrix(Float32 saturation, const SkColor4f &tint) noexcept
168 {
169 constexpr Float32 w[3] { 0.2126f, 0.7152f, 0.0722f };
170 const Float32 a { std::clamp(tint.fA, 0.f, 1.f) };
171 const Float32 t[3] { tint.fR, tint.fG, tint.fB };
172 std::array<Float32, 12> m {};
173
174 for (int i = 0; i < 3; i++)
175 {
176 for (int j = 0; j < 3; j++)
177 m[i * 4 + j] = (1.f - a) * ((1.f - saturation) * w[j] + (i == j ? saturation : 0.f));
178
179 m[i * 4 + 3] = a * t[i];
180 }
181
182 return m;
183 }
