73 bool setImage(std::shared_ptr<RImage> image) noexcept
74 {
75 if (image.get() == m_image.get())
76 return false;
77
78 /* Swapping between images is covered by the fine-grained damage the owner adds (e.g. a
79 * client's buffer damage), but gaining or losing the image changes whether the node draws
80 * AT ALL. That transition isn't covered by such damage: it may have been consumed by a
81 * frame where the node couldn't draw yet (null image), leaving the node permanently stale
82 * on the buffers composited before the image arrived. */
83 if ((m_image == nullptr) != (image == nullptr))
84 addDamage(AK_IRECT_INF);
85
86 addChange(CHImage);
87 m_image = image;
88 return true;
89 }
