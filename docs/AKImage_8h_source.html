69 bool setImage(std::shared_ptr<RImage> image) noexcept
70 {
71 if (image.get() == m_image.get())
72 return false;
73
74 /* Swapping between images is covered by the fine-grained damage the owner adds (e.g. a
75 * client's buffer damage), but gaining or losing the image changes whether the node draws
76 * AT ALL. That transition isn't covered by such damage: it may have been consumed by a
77 * frame where the node couldn't draw yet (null image), leaving the node permanently stale
78 * on the buffers composited before the image arrived. */
79 if ((m_image == nullptr) != (image == nullptr))
80 addDamage(AK_IRECT_INF);
81
82 addChange(CHImage);
83 m_image = image;
84 return true;
85 }
