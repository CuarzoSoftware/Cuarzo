99 /*
100 * Kinetic scrolling, per axis (0 = X, 1 = Y), like macOS:
101 *
102 * - Tracking: the fingers move the content 1:1; past an edge it's rubber-banded.
103 * - Momentum: after lifting them, it keeps going and slows down exponentially.
104 * - Spring: past an edge (released there or reached by momentum), a critically damped spring
105 * brings it back. Its overshoot is rubber-banded too (never past the limit).
106 * - Wheel: a discrete wheel step glides to its target.
107 */
108 struct Axis
109 {
110 enum class Phase { Idle, Tracking, Momentum, Spring, Wheel };
111 Phase phase { Phase::Idle };
112 SkScalar vel { 0.f }; // Logical px per ms
113 SkScalar raw { 0.f }; // Tracking: the offset without the rubber band
114 SkScalar overshoot { 0.f }; // Spring: distance past the edge without the rubber band
115 SkScalar edge { 0.f }; // Spring: the edge it springs back to
116 SkScalar wheelTarget { 0.f };
117 UInt64 lastEventUs { 0 };
118 };
119
120 Axis m_axes[2];
121 CZLinearAnimation m_ticker; // Drives the axes not Idle nor Tracking
122 UInt64 m_lastTickUs { 0 };
123
