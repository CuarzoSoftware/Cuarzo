18template<typename T, typename... Rest>
19inline void HashCombine(std::size_t &seed, const T &v, const Rest &... rest)
20{
21 std::hash<T> hasher;
22 seed ^= hasher(v) + 0x9e3779b9 + (seed << 6) + (seed >> 2);
23 HashCombine(seed, rest...);
24}

63
69 std::shared_ptr<T> get(Args... args)
70 {
71 assert(m_factory && "Missing factory function");
72
73 ++m_accessCount;
74
75 // Occasionally clean expired
76 if (m_accessCount % CleanupInterval == 0)
77 cleanupExpired();
78
79 Key key(args...);
80
81 auto it = m_cache.find(key);
82
83 if (it != m_cache.end())
84 {
85 if (auto existing = it->second.lock())
86 return existing;
87 else
88 m_cache.erase(it);
89 }
90
91 // Create new asset
92 auto asset = m_factory(args...);
93
94 if (!asset) return {};
95
96 m_cache[key] = asset;
97 return asset;
98 }
99
100private:
101
102 void cleanupExpired()
103 {
104 for (auto it = m_cache.begin(); it != m_cache.end();)
105 {
106 if (it->second.expired())
107 it = m_cache.erase(it);
108 else
109 ++it;
110 }
111 }
112 std::unordered_map<Key, std::weak_ptr<T>, CZTupleHash<Args...>> m_cache;
113 FactoryFunc m_factory;
114 size_t m_accessCount {};
115 static constexpr size_t CleanupInterval = 100;
116};
