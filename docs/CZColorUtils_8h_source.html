67 static constexpr SkColor SrcOver(SkColor A, SkColor B) noexcept
68 {
69 // Extract unpremultiplied 0–255 channels
70 Float32 Ar = U8ToFloat32(SkColorGetR(A));
71 Float32 Ag = U8ToFloat32(SkColorGetG(A));
72 Float32 Ab = U8ToFloat32(SkColorGetB(A));
73 Float32 Aa = U8ToFloat32(SkColorGetA(A));
74
75 Float32 Br = U8ToFloat32(SkColorGetR(B));
76 Float32 Bg = U8ToFloat32(SkColorGetG(B));
77 Float32 Bb = U8ToFloat32(SkColorGetB(B));
78 Float32 Ba = U8ToFloat32(SkColorGetA(B));
79
80 // Convert to premultiplied
81 Float32 Apr = Ar * Aa;
82 Float32 Apg = Ag * Aa;
83 Float32 Apb = Ab * Aa;
84
85 Float32 Bpr = Br * Ba;
86 Float32 Bpg = Bg * Ba;
87 Float32 Bpb = Bb * Ba;
88
89 // ---- kSrcOver : B over A ----
90 Float32 outA = Ba + Aa * (1.0f - Ba);
91 Float32 outPr = Bpr + Apr * (1.0f - Ba);
92 Float32 outPg = Bpg + Apg * (1.0f - Ba);
93 Float32 outPb = Bpb + Apb * (1.0f - Ba);
94
95 // Convert back to unpremultiplied
96 Float32 invA = (outA > 0.0f) ? (1.0f / outA) : 0.0f;
97
98 UInt8 r = Float32ToU8(outPr * invA);
99 UInt8 g = Float32ToU8(outPg * invA);
100 UInt8 b = Float32ToU8(outPb * invA);
101 UInt8 a = Float32ToU8(outA);
102
103 return SkColorSetARGB(a, r, g, b);
104 }
